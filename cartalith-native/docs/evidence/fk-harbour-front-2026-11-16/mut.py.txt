sel = sys.argv[1:]
def h(p): return hashlib.sha256(open(p, 'rb').read()).hexdigest()
files = sorted({m[1] for m in M})
before = {f: h(f) for f in files}
res = []
env = dict(os.environ, CARGO_TARGET_DIR='C:/cw/rulingfk_target')
for mid, f, old, new, crate, filt in M:
    if sel and not any(mid.startswith(s + ' ') for s in sel):
        continue
    raw = open(f, 'rb').read()
    txt = raw.decode('utf-8')
    crlf = '\r\n' in txt
    t = txt.replace('\r\n', '\n')
    n = t.count(old)
    assert n == 1, (mid, n)
    try:
        m2 = t.replace(old, new)
        if crlf: m2 = m2.replace('\n', '\r\n')
        open(f, 'wb').write(m2.encode('utf-8'))
        r = subprocess.run(['cargo', 'test', '-p', crate, '--lib', filt], cwd='C:/cw/rulingfk/cartalith-native', env=env, capture_output=True, text=True, encoding='utf-8', errors='replace')
        out = r.stdout + r.stderr
        if 'could not compile' in out or 'error[' in out:
            v = 'BUILD_ERROR'
        elif r.returncode != 0:
            v = 'KILLED'
        else:
            v = 'SURVIVED'
        fails = [l for l in out.splitlines() if l.startswith('test ') and 'FAILED' in l]
        res.append((mid, v, fails[:3]))
        print(mid, v, fails[:3], flush=True)
    finally:
        open(f, 'wb').write(raw)
for f in files:
    assert h(f) == before[f], ('NOT RESTORED', f)
print('restored, hashes match')
