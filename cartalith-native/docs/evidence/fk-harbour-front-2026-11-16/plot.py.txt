"""plot.py <json> <out.png>: a quick plan of one town (open water blue, buildings brown, edges by class, ring grey, river, market red, pin black)."""
import json, sys
from PIL import Image, ImageDraw
d = json.load(open(sys.argv[1]))
S = 0.6
W, H = int(1700 * S), int(1250 * S)
im = Image.new("RGB", (W, H), (226, 222, 205))
dr = ImageDraw.Draw(im)
c = d["cell"] or 22.0
mw = d["mw"]
for k, v in enumerate(d["open"]):
    if v:
        i, j = k % mw, k // mw
        dr.rectangle([i * c * S, j * c * S, (i + 1) * c * S, (j + 1) * c * S], fill=(110, 150, 200))
P = lambda p: (p[0] * S, p[1] * S)
if len(d["river"]) >= 2:
    dr.line([P(p) for p in d["river"]], fill=(60, 100, 190), width=max(1, int(d["river_w"] * S)))
col = {"primary": (200, 120, 40), "street": (120, 120, 120), "lane": (170, 170, 170), "quay": (20, 20, 120), "ringroad": (90, 90, 90)}
for e in d["edges"]:
    dr.line([P(e[2:4]), P(e[4:6])], fill=col.get(e[0], (0, 0, 0)), width=max(1, int(e[1] * S)))
for b in d["buildings"]:
    if len(b) >= 3:
        dr.polygon([P(p) for p in b], fill=(140, 70, 50))
if len(d["ring"]) >= 3:
    dr.line([P(p) for p in d["ring"] + d["ring"][:1]], fill=(80, 80, 80), width=3)
for key, colr in (("market", (220, 0, 0)), ("pin", (0, 0, 0))):
    x, y = P(d[key]); dr.ellipse([x - 5, y - 5, x + 5, y + 5], outline=colr, width=3)
dr.rectangle([0, 0, W, 14], fill=(0, 0, 0))
dr.text((3, 1), f"{d['seed']} #{d['i']} ({d['x']},{d['y']}) {d['kind']} {d['row'][:150]}", fill=(255, 255, 255))
im.save(sys.argv[2])
