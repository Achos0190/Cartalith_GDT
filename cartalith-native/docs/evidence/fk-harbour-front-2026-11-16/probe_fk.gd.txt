extends Node
## SCRATCH COPY for Ruling FK frames (rulingfk worktree, never committed; from the FI copy): --fk 0|1
## (urban_harbour_front and civ.port_reach), --list 1 (print every settlement and quit), --cells x:y,x:y (frame the
## settlement nearest each cell: map crops at --zoom and --zoom2, and the City Viewer), --pts x:y,x:y (map crops
## centred on raw cells), --tag T (file name tag).
## The owner's town-on-its-river report (2026-11-14: *"The river passes below it. It's not on the river. And for the
## same matter double check this on the bigger map that it sits correctly."*): frames of the real shell, the same
## settlement in the City Viewer and on the world map, for `docs/evidence/town-river-2026-11-14/`.
##
##   Godot_v4.7.2-stable_win64_console.exe --path . --resolution 1600x1000 _townriver_probe.tscn -- --out <dir> [--seed N] [--viewer I] [--crops I,J,K] [--zoom Z] [--front 0|1] [--grid WxH] [--min-pop P]
##
## **Windowed, never `--headless`** (it reads rendered pixels). Defaults: seed 483920 at 384 x 288 over 800 km (`--grid` for another, e.g. the app's 2048 x 1311) (the
## grid `_wallhug_probe.gd` sent the owner's frame from), City Viewer for settlement 6 in the map look, world-map
## crops of settlement 6 at x16.
##
## What it checks (each names itself): the world generated, the City Viewer frame drew (many distinct colours), and
## each map crop drew. The comparison it exists for is by eye and by `town_river_census` (Rust), which measures the
## same towns: the frames are the evidence a number cannot give.
## Frames: `<out>/viewer_<seed>_<i>.png`, `<out>/map_<seed>_<i>_x<z>.png`.
## Result: `RESULT: PASS|FAIL (<n> failed) checks=<n>`; exit 0 pass, 1 fail, 2 bad argument or watchdog.

var _out := ""
var _vp: SubViewport
var _app: Node
var _fails := 0
var _checks := 0

func _ok(cond: bool, what: String) -> void:
	_checks += 1
	print(("  PASS  %s" if cond else "  FAIL  %s") % what)
	if not cond:
		_fails += 1

func _frames(n: int) -> void:
	for i in n:
		await get_tree().process_frame

func _distinct(img: Image) -> int:
	var seen := {}
	for y in range(0, img.get_height(), 3):
		for x in range(0, img.get_width(), 3):
			var c := img.get_pixel(x, y)
			seen[Vector3i(int(c.r * 15.0), int(c.g * 15.0), int(c.b * 15.0))] = true
	return seen.size()

func _shot(rect: Rect2) -> Image:
	await RenderingServer.frame_post_draw
	var img: Image = _vp.get_texture().get_image()
	return img.get_region(Rect2i(rect).intersection(Rect2i(Vector2i.ZERO, img.get_size())))

func _ready() -> void:
	get_tree().create_timer(2400.0).timeout.connect(func() -> void:
		push_error("townriver probe watchdog")
		get_tree().quit(2))
	if DisplayServer.get_name() == "headless":
		push_error("_townriver_probe must be windowed")
		get_tree().quit(2)
		return
	var seed_n := 483920
	var viewer := 6
	var crops: Array[int] = [6]
	var zoom := 16.0
	var front := -1
	var gw := 384
	var gh := 288
	## `--min-pop P`: list every town of at least P people (default 3 000; 0 lists every settlement).
	var min_pop := 3000
	var fk := -1
	var zoom2 := 0.0
	var pts: Array[Vector2i] = []
	var list_only := false
	var cells: Array[Vector2i] = []
	var tag := "x"
	var args := OS.get_cmdline_user_args()
	var i := 0
	while i < args.size():
		match args[i]:
			"--out":
				_out = args[i + 1]
			"--seed":
				seed_n = int(args[i + 1])
			"--viewer":
				viewer = int(args[i + 1])
			"--crops":
				crops.clear()
				for t in String(args[i + 1]).split(",", false):
					crops.append(int(t))
			"--zoom":
				zoom = float(args[i + 1])
			"--front":
				front = int(args[i + 1])
			"--fk":
				fk = int(args[i + 1])
			"--zoom2":
				zoom2 = float(args[i + 1])
			"--pts":
				for t in String(args[i + 1]).split(",", false):
					var pq := t.split(":")
					pts.append(Vector2i(int(pq[0]), int(pq[1])))
			"--list":
				list_only = int(args[i + 1]) == 1
			"--tag":
				tag = args[i + 1]
			"--cells":
				for t in String(args[i + 1]).split(",", false):
					var xy := t.split(":")
					cells.append(Vector2i(int(xy[0]), int(xy[1])))
			"--min-pop":
				min_pop = int(args[i + 1])
			"--grid":
				var wh := String(args[i + 1]).split("x")
				gw = int(wh[0])
				gh = int(wh[1])
			_:
				push_error("unknown argument " + args[i])
				get_tree().quit(2)
				return
		i += 2
	if _out == "":
		push_error("--out <dir> is required")
		get_tree().quit(2)
		return
	DirAccess.make_dir_recursive_absolute(_out)
	_vp = SubViewport.new()
	_vp.size = Vector2i(1600, 1000)
	_vp.gui_embed_subwindows = true
	_vp.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	add_child(_vp)
	_app = load("res://shell/app.tscn").instantiate()
	_vp.add_child(_app)
	await get_tree().create_timer(1.0).timeout
	var bridge = _app.bridge
	## `--front 0|1` sets Ruling EX's river frontage term (`civ.river_front`) before the world is made: 0 keeps the
	## placements the owner's frame came from, 1 is the shipped app. Unset leaves the app's own default. An engine
	## without the parameter (the before build) ignores it.
	if front >= 0 and not bridge.param_info("civ.river_front").is_empty():
		bridge.param_set("civ.river_front", front == 1)
	if fk >= 0:
		bridge.param_set("urban_harbour_front", fk == 1)
		bridge.param_set("civ.port_reach", fk == 1)
	print("  fk %d: urban_harbour_front %s port_reach %s port_shore %s" % [fk, bridge.param_get("urban_harbour_front"), bridge.param_get("civ.port_reach"), bridge.param_get("civ.port_shore")])
	bridge.generate({"seed": seed_n, "width_km": 800.0, "grid_w": gw, "grid_h": gh, "archetype": "", "villages": true})
	while bridge.generating:
		await get_tree().create_timer(0.25).timeout
	await get_tree().create_timer(1.0).timeout
	if _app.open_project_dialog:
		_app.open_project_dialog.hide()
	await _frames(6)
	var places: Array = bridge.settlements()
	_ok(places.size() > maxi(viewer, 0), "the world generated (%d settlements)" % places.size())
	if list_only:
		for k in places.size():
			var lt0: Dictionary = bridge.urban_layouts(PackedInt32Array([k]))[0]
			var rv0: PackedVector2Array = lt0.get("river", PackedVector2Array())
			print("S %d %d %d %d %s %s %s market %s pin %s river_end %s" % [k, int(places[k]["x"]), int(places[k]["y"]), int(places[k]["population"]), places[k]["kind"], places[k]["coastal"], lt0.get("site_kind", "?"), lt0.get("market", "?"), lt0.get("pin", "-"), rv0[rv0.size() - 1] if rv0.size() > 0 else "-"])
		## Every drawn road's points (for choosing the roads whose route changed between --fi 0 and 1).
		var roads: Array = bridge.world_gen.get_roads()
		for r in roads:
			var pp: PackedVector2Array = r.get("points", PackedVector2Array())
			var parts := PackedStringArray()
			for q in pp:
				parts.append("%.2f,%.2f" % [q.x, q.y])
			print("R %s %s" % [r.get("way_type", "?"), " ".join(parts)])
		_finish()
		return
	if not cells.is_empty():
		crops.clear()
		for c in cells:
			var best := -1
			var bd := INF
			for k in places.size():
				var d := Vector2(float(places[k]["x"]) - c.x, float(places[k]["y"]) - c.y).length()
				if d < bd:
					bd = d
					best = k
			crops.append(best)
			print("  cell %s -> settlement #%d at (%d,%d), %.2f cells off, coastal %s" % [c, best, int(places[best]["x"]), int(places[best]["y"]), bd, places[best]["coastal"]])

	## Every town of at least `min_pop` people, with what its layout was given: the list a frame is chosen from.
	for k in places.size():
		if int(places[k].get("population", 0)) >= min_pop:
			var lt: Dictionary = bridge.urban_layouts(PackedInt32Array([k]))[0]
			## Market to the layout's river, in metres (`Geometry2D` on each segment); `-` without a river of two points.
			var rv: PackedVector2Array = lt.get("river", PackedVector2Array())
			var mk: Vector2 = lt.get("market", Vector2.ZERO)
			var dmin := INF
			for j in rv.size() - 1:
				dmin = minf(dmin, mk.distance_to(Geometry2D.get_closest_point_to_segment(mk, rv[j], rv[j + 1])))
			print("  town #%d cell (%d,%d) pop %d site_kind %s uses_real_water %s market-to-river %s m" % [k,
				int(places[k]["x"]), int(places[k]["y"]), int(places[k]["population"]), lt.get("site_kind", "?"),
				lt.get("uses_real_water", "?"), ("%.0f" % dmin) if dmin < INF else "-"])

	var host: Control = _app.viewport
	var ov: Control = host.overlay
	## The region names are cleared for the crops: at x16 a region label is larger than the town and covers it. The
	## crops are taken before the City Viewer opens, so no selection ring is drawn round the marker either.
	ov.set_labels([])
	var zooms: Array[float] = [zoom]
	if zoom2 > 0.0:
		zooms.append(zoom2)
	for kz in crops.size() * zooms.size():
		var k: int = crops[kz / zooms.size()]
		zoom = zooms[kz % zooms.size()]
		if k < 0 or k >= places.size():
			continue
		var lk: Dictionary = bridge.urban_layouts(PackedInt32Array([k]))[0]
		print("\n== world map, settlement #%d at x%d (cell %d,%d, site_kind %s, uses_real_water %s) ==" % [k, int(zoom),
			int(places[k]["x"]), int(places[k]["y"]), lk.get("site_kind", "?"), lk.get("uses_real_water", "?")])
		host.reset_view()
		await _frames(6)
		var s: Dictionary = places[k]
		## `_cell_to_screen` is where the overlay draws the marker (the cell centre), in the overlay's own space.
		var at_ov: Vector2 = ov._cell_to_screen(Vector2(float(s["x"]), float(s["y"])), ov._displayed_rect())
		var at_global: Vector2 = ov.get_global_transform_with_canvas() * at_ov
		var local: Vector2 = host.get_global_transform_with_canvas().affine_inverse() * at_global
		host._zoom_at(local, zoom / host.zoom())
		await _frames(4)
		## Then pan so the marker sits at the middle of the map view, clear of the bars at its edges.
		var at1: Vector2 = host.get_global_transform_with_canvas().affine_inverse() * (ov.get_global_transform_with_canvas() * at_ov)
		host._camera.position += host.size * 0.5 - at1
		await get_tree().create_timer(6.0).timeout
		await _frames(10)
		var at2: Vector2 = ov.get_global_transform_with_canvas() * at_ov
		var im2 := await _shot(Rect2(at2 - Vector2(300, 300), Vector2(600, 600)))
		im2.save_png(_out.path_join("map_%s_%d_%d_%d_x%d.png" % [tag, seed_n, int(s["x"]), int(s["y"]), int(zoom)]))
		print("  marker at %s (cell %d,%d), zoom %.1f" % [at2, int(s["x"]), int(s["y"]), host.zoom()])
		_ok(_distinct(im2) > 20, "the map crop of #%d drew (%d distinct colours)" % [k, _distinct(im2)])

	## Raw-cell crops (roads whose route changed): the same zoom and centring, no settlement.
	for c in pts:
		host.reset_view()
		await _frames(6)
		var p_ov: Vector2 = ov._cell_to_screen(Vector2(float(c.x), float(c.y)), ov._displayed_rect())
		var p_loc: Vector2 = host.get_global_transform_with_canvas().affine_inverse() * (ov.get_global_transform_with_canvas() * p_ov)
		host._zoom_at(p_loc, zoom / host.zoom())
		await _frames(4)
		var p1: Vector2 = host.get_global_transform_with_canvas().affine_inverse() * (ov.get_global_transform_with_canvas() * p_ov)
		host._camera.position += host.size * 0.5 - p1
		await get_tree().create_timer(6.0).timeout
		await _frames(10)
		var p2: Vector2 = ov.get_global_transform_with_canvas() * p_ov
		var imp := await _shot(Rect2(p2 - Vector2(300, 300), Vector2(600, 600)))
		imp.save_png(_out.path_join("pt_%s_%d_%d_%d_x%d.png" % [tag, seed_n, c.x, c.y, int(zoom)]))
		_ok(_distinct(imp) > 20, "the map crop at cell %s drew" % c)
	if list_only == false and cells.is_empty() and not pts.is_empty():
		_finish()
		return

	var viewers: Array[int] = [viewer]
	if not cells.is_empty():
		viewers = crops
	for vw in viewers:
		viewer = vw
		if viewer < 0 or viewer >= places.size():
			continue
		print("\n== City Viewer, settlement #%d (map look) ==" % viewer)
		## What the layout was given: its site kind, whether its water is the map's own, and its river.
		var lay: Dictionary = bridge.urban_layouts(PackedInt32Array([viewer]))[0]
		var riv: PackedVector2Array = lay.get("river", PackedVector2Array())
		print("  cell (%d,%d) site_kind %s uses_real_water %s river %d pts %s..%s market %s pin %s" % [
			int(places[viewer]["x"]), int(places[viewer]["y"]), lay.get("site_kind", "?"), lay.get("uses_real_water", "?"),
			riv.size(), riv[0] if riv.size() > 0 else Vector2.INF, riv[riv.size() - 1] if riv.size() > 0 else Vector2.INF,
			lay.get("market", "?"), lay.get("pin", "-")])
		## The water the layout draws: river width, the water polygon, the mask runs, and its crossings.
		print("  river_w %s water_poly %d pts water_mask_runs %d bridges %d ford %s" % [lay.get("river_w", "?"),
			PackedVector2Array(lay.get("water_poly", PackedVector2Array())).size(), Array(lay.get("water_mask_runs", [])).size(),
			PackedVector2Array(lay.get("bridges", PackedVector2Array())).size(), lay.has("ford")])
		## The water mask's extent: its area (m^2) and x/y range in the layout's frame, and the market.
		var w_area := 0.0
		var w_lo := Vector2(INF, INF)
		var w_hi := Vector2(-INF, -INF)
		for run in lay.get("water_mask_runs", []):
			var rp: PackedVector2Array = run
			w_area += absf((rp[2].x - rp[0].x) * (rp[2].y - rp[0].y))
			for q in rp:
				w_lo = Vector2(minf(w_lo.x, q.x), minf(w_lo.y, q.y))
				w_hi = Vector2(maxf(w_hi.x, q.x), maxf(w_hi.y, q.y))
		print("  water mask area %.0f m2 extent %s..%s" % [w_area, w_lo, w_hi])
		var shore_b: Dictionary = preload("res://shell/urban_shore.gd").cached(lay.duplicate(), [])
		var sr: Dictionary = shore_b.get("rings", {})
		print("  shore: rings %d smooth %d pieces %d mesh %s" % [Array(sr.get("raw", [])).size(), Array(sr.get("smooth", [])).size(), Array(shore_b.get("pieces", [])).size(), shore_b.get("mesh") != null])
		for pc in shore_b.get("pieces", []):
			var pcb: PackedVector2Array = pc
			var pr := Rect2(pcb[0], Vector2.ZERO)
			var parea := 0.0
			for k in pcb.size():
				pr = pr.expand(pcb[k])
				var q0: Vector2 = pcb[k]
				var q1: Vector2 = pcb[(k + 1) % pcb.size()]
				parea += q0.x * q1.y - q1.x * q0.y
			print("    piece %d pts tri %d bounds %s area %.0f m2" % [pcb.size(), Geometry2D.triangulate_polygon(pcb).size(), pr, absf(parea) * 0.5])
		## The layout as data, for a plan plot (`plot_gd.py`): points as [x, y] pairs.
		var pv := func(a) -> Array:
			var o: Array = []
			for q in a:
				o.append([snappedf(q.x, 0.1), snappedf(q.y, 0.1)])
			return o
		var dump := {"market": pv.call([lay.get("market", Vector2.ZERO)]), "pin": pv.call([lay.get("pin", lay.get("market", Vector2.ZERO))]),
			"river": pv.call(lay.get("river", PackedVector2Array())), "river_w": lay.get("river_w", 0.0),
			"ring": pv.call(lay.get("wall_ring", PackedVector2Array())), "pop": lay.get("pop", 0), "pop_target": lay.get("pop_target", 0),
			"max_rf": lay.get("max_rf_m", 0.0), "buildings": [], "water": [], "streets": {}}
		for b in lay.get("buildings", []):
			dump["buildings"].append(pv.call(b))
		for r in lay.get("water_mask_runs", []):
			dump["water"].append(pv.call(r))
		var st: Dictionary = lay.get("streets", {})
		for k in st.keys():
			dump["streets"][k] = pv.call(st[k])
		var fdump := FileAccess.open(_out.path_join("layout_%s_%d_%d_%d.json" % [tag, seed_n, int(places[viewer]["x"]), int(places[viewer]["y"])]), FileAccess.WRITE)
		fdump.store_string(JSON.stringify(dump))
		fdump.close()
		print("  layout: buildings %d pop %s of %s ring %d pts market %s pin %s" % [dump["buildings"].size(), dump["pop"], dump["pop_target"], dump["ring"].size(), dump["market"], dump["pin"]])
		_app.open_city_viewer(viewer)
		await _frames(12)
		var cv = _app.city_viewer_window
		cv.size = Vector2i(1560, 960)
		cv.position = Vector2i(20, 20)
		await _frames(8)
		cv._look_picker.select(0)
		cv._look_picker.item_selected.emit(0)
		await _frames(3)
		cv._reset_view()
		await _frames(6)
		var rect := Rect2(cv._canvas.get_global_transform_with_canvas().origin + Vector2(cv.position), cv._canvas.size)
		var im := await _shot(rect)
		im.save_png(_out.path_join("viewer_%s_%d_%d_%d.png" % [tag, seed_n, int(places[viewer]["x"]), int(places[viewer]["y"])]))
		_ok(_distinct(im) > 40, "the City Viewer frame drew (%d distinct colours)" % _distinct(im))
		var vtf: Dictionary = cv._view_transform()
		print("  viewer transform scale %.4f origin %s; mask corners on canvas %s %s; canvas %s" % [float(vtf["scale"]), vtf["origin"],
			Vector2(vtf["origin"]) + w_lo * float(vtf["scale"]), Vector2(vtf["origin"]) + w_hi * float(vtf["scale"]), cv._canvas.size])
		## The same view zoomed out (x0.25, the viewer's floor): the whole 1.7 km town box, so a thin sea at the box edge shows.
		cv._zoom_by(0.25, cv._canvas.size * 0.5)
		await _frames(6)
		var imz := await _shot(rect)
		imz.save_png(_out.path_join("viewerz_%s_%d_%d_%d.png" % [tag, seed_n, int(places[viewer]["x"]), int(places[viewer]["y"])]))
		cv.hide()
		await _frames(4)

	_finish()

func _finish() -> void:
	print("RESULT: %s (%d failed) checks=%d" % ["PASS" if _fails == 0 else "FAIL", _fails, _checks])
	get_tree().quit(0 if _fails == 0 else 1)
