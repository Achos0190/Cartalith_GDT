# Ruling FI mutants: exact literal replace (occurs exactly once), run the named tests, restore in finally.
import subprocess, sys, hashlib, os
ROOT = 'C:/cw/rulingfi/cartalith-native'
CIV = ROOT + '/crates/cartalith-civ/src/lib.rs'
GOD = ROOT + '/crates/cartalith-godot/src/lib.rs'
CIV_T = ['cargo', 'test', '-p', 'cartalith-civ', '--lib', '-j', '4', '--', 'shore', 'every_drawn_river', 'port_candidate']
GOD_T = ['cargo', 'test', '-p', 'cartalith-godot', '--lib', '-j', '4', '--', 'port_candidates_move_onto_the_shore', 'every_drawn_river_crossing_costs']
M = [
 ('M1 nudge a town the radius test did not badge', CIV, 'let (fx, fy) = if shore && coastal {', 'let (fx, fy) = if shore {', CIV_T),
 ('M2 no river preference', CIV, 'let off = keep_river.is_some_and(|on| !on[i]);', 'let off = false;', CIV_T),
 ('M3 skip ok()', CIV, 'if better && ok(ux, uy) && has_sea(ux, uy) {', 'if better && has_sea(ux, uy) {', CIV_T),
 ('M4 ignore the reach circle', CIV, 'if d2 == 0 || d2 > reach * reach || xx < 0', 'if d2 == 0 || xx < 0', CIV_T),
 ('M5 suitability before distance', CIV, 'Some((_, _, bo, bd, bs)) => (off, d2) < (bo, bd) || ((off, d2) == (bo, bd) && sv > bs),', 'Some((_, _, bo, _bd, bs)) => (!off && bo) || (off == bo && sv > bs),', CIV_T),
 ('M6 badge not re-taken at the new cell', CIV, '                        coastal = civ_is_coastal(nx, ny, coast_r, true, field, Some(wb), gw, gh, sea, world);\n', '', CIV_T),
 ('M7 map crossings branch never taken', CIV, '(true, Some(fl), Some(ro)) => civ_map_river_crossing_cost(fl, flow_thresh, ro, gw, gh, xy, wh),', '(false, Some(fl), Some(ro)) => civ_map_river_crossing_cost(fl, flow_thresh, ro, gw, gh, xy, wh),', CIV_T),
 ('M8 no threshold floor', CIV, 'let mag = river_class::magnitude_raw(q.max(flow_thresh), flow_thresh);', 'let mag = river_class::magnitude_raw(q, flow_thresh);', CIV_T),
 ('M9 only the sampled cell', CIV, '        for fx in routing_block(x, w, gw) {\n            let i = fy * gw + fx;', '        for fx in [x * gw / w] {\n            let i = fy * gw + fx;', CIV_T),
 ('M10 max(sampled) dropped', CIV, 'drawn.map_or(sampled, |d| d.max(sampled))', 'drawn.map_or(sampled, |d| d)', CIV_T),
 ('M11 placement passes false', GOD, '                opts.port_shore,\n', '                false,\n', GOD_T),
 ('M12 network passes false', GOD, '        &mut placements, passes, gw, gh, sea_level, &ws.field, &ws.flow_discharge, &river_order, &biome, &wb.classification, world, map_width_km,\n        opts.map_crossings,', '        &mut placements, passes, gw, gh, sea_level, &ws.field, &ws.flow_discharge, &river_order, &biome, &wb.classification, world, map_width_km,\n        false,', GOD_T),
]
env = dict(os.environ, CARGO_TARGET_DIR='C:/cw/rulingfi_target')
only = sys.argv[1:]
for name, path, old, new, cmd in M:
    if only and name.split()[0] not in only:
        continue
    src = open(path, encoding='utf-8', newline='').read()
    h = hashlib.sha256(src.encode()).hexdigest()
    assert src.count(old) == 1, (name, src.count(old))
    try:
        open(path, 'w', encoding='utf-8', newline='').write(src.replace(old, new))
        r = subprocess.run(cmd, cwd=ROOT, env=env, capture_output=True, text=True, encoding='utf-8', errors='replace')
        out = r.stdout + r.stderr
        if 'error[' in out or 'could not compile' in out:
            verdict = 'BUILD_ERROR'
        elif 'test result: FAILED' in out or 'FAILED' in out:
            verdict = 'KILLED'
        elif 'test result: ok' in out:
            verdict = 'SURVIVED'
        else:
            verdict = 'UNKNOWN'
        fails = [l for l in out.splitlines() if l.startswith('test ') and 'FAILED' in l]
        print(f'{name}: {verdict} {fails}', flush=True)
    finally:
        open(path, 'w', encoding='utf-8', newline='').write(src)
        assert hashlib.sha256(open(path, encoding='utf-8', newline='').read().encode()).hexdigest() == h
print('MUT_DONE')
