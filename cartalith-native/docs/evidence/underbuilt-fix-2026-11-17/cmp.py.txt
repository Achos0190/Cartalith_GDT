import re,sys
def parse(p):
    out={};seed=None;lines=open(p,encoding='utf-8',errors='replace').read().split('\n')
    pend=[]
    for l in lines:
        m=re.search(r'#\s*(\d+) off:.*\| on:\s+(\w+)\s+bld\s+(\d+) housed\s+(\d+)/\s*(\d+) \(\s*([\d.]+) %\)',l)
        if m: pend.append((int(m.group(1)),m.group(2),int(m.group(3)),int(m.group(4)),int(m.group(5)),float(m.group(6))))
        m=re.match(r'seed (\d+) at',l)
        if m:
            out[int(m.group(1))]=pend;pend=[]
    return out
a=parse(sys.argv[1]);b=parse(sys.argv[2])
for sd in a:
    for x,y in zip(a[sd],b[sd]):
        if x!=y: print(sd,x[0],x[1],'tgt',x[4],'bld',x[2],'->',y[2],'housed%',x[5],'->',y[5], 'WORSE' if y[5]<x[5]-0.05 and x[4]>=1500 else '')
