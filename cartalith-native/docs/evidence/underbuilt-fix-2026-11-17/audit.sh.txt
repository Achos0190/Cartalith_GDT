#!/bin/bash
# audit.sh <exe> <outdir> <sizes> : the five town audits, seeds 483920 912345 246371, 800 km, villages on, app defaults (+ EXTRA_ENV)
EXE=$1; OUT=$2; mkdir -p $OUT
cd /c/cw/gb/cartalith-native/crates/cartalith-godot
for sz in $3; do
  gw=${sz%%:*}; gh=${sz##*:}
  for a in both_banks_audit town_housing_audit riverbank_gap_audit harbour_front_audit stretch_water_audit; do
    env TOWN_RIVER_GW=$gw TOWN_RIVER_GH=$gh TOWN_RIVER_VILLAGES=1 TOWN_RIVER_LIST=1 ${EXTRA_ENV:-} $EXE $a --ignored --nocapture --test-threads 1 > $OUT/${a}_$gw.txt 2>&1 &
  done
  wait
  echo "$gw done $(date +%T)"
done
echo ALLDONE
