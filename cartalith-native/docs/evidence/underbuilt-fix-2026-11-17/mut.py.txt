base={f:open(R+f,encoding='utf-8').read() for f in set(m[0] for m in M)}
hashes={f:hashlib.md5(v.encode()).hexdigest() for f,v in base.items()}
out=[]
try:
    for f,a,b,name in M:
        s=base[f]
        assert s.count(a)==1,(name,s.count(a))
        open(R+f,'w',encoding='utf-8').write(s.replace(a,b))
        try:
            r=subprocess.run(["cargo","test","-p","cartalith-urban","--lib","--","water_widen","bridge_search","water_sweep","widened_reach","disc_land","a_town_bridge","big_river_town","far_bank"],capture_output=True,text=True,env=env,cwd="C:/cw/gb/cartalith-native")
            txt=r.stdout+r.stderr
            if "could not compile" in txt: st="BUILD_ERROR"
            elif r.returncode!=0: st="KILLED"
            else: st="SURVIVED"
        finally:
            open(R+f,'w',encoding='utf-8').write(s)
        out.append(f"{st:12} {f:14} {name}")
        print(out[-1],flush=True)
finally:
    for f,v in base.items(): open(R+f,'w',encoding='utf-8').write(v)
    for f,v in base.items(): assert hashlib.md5(open(R+f,encoding='utf-8').read().encode()).hexdigest()==hashes[f]
open("C:/cw/gb_ev/mutants.txt","w").write("\n".join(out)+"\n")
