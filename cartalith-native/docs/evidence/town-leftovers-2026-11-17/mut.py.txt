for name, f, old, new, cmd in M:
    if only and not any(name.startswith(o) for o in only):
        continue
    src = open(f, 'rb').read()
    h = hashlib.sha256(src).hexdigest()
    s = src.decode('utf-8')
    assert s.count(old) == 1, (name, s.count(old))
    try:
        open(f, 'wb').write(s.replace(old, new).encode('utf-8'))
        p = subprocess.run(cmd, cwd='C:/cw/town2/cartalith-native', env=env, capture_output=True, text=True, encoding='utf-8', errors='replace')
        out = p.stdout + p.stderr
        if 'error[' in out or 'could not compile' in out:
            v = 'BUILD_ERROR'
        elif p.returncode != 0:
            v = 'KILLED'
        else:
            v = 'SURVIVED'
        fails = [l for l in out.splitlines() if l.startswith('test ') and l.endswith('FAILED')]
        print(f"{v:11s} {name}  {'; '.join(x.split()[1] for x in fails)[:200]}", flush=True)
    finally:
        open(f, 'wb').write(src)
        assert hashlib.sha256(open(f, 'rb').read()).hexdigest() == h
