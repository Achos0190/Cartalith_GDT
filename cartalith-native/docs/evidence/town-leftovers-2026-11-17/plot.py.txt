"""plot.py <json> <out.png> [label]: plan of one both_banks_audit town JSON (river, edges by class, buildings, parcels, ring, plaza, market)."""
import json, sys
from PIL import Image, ImageDraw
d = json.load(open(sys.argv[1]))
S = 0.6
W, H = int(1700 * S), int(1250 * S)
im = Image.new("RGB", (W, H), (226, 222, 205))
dr = ImageDraw.Draw(im)
P = lambda p: (p[0] * S, p[1] * S)
if len(d["water_poly"]) >= 3:
    dr.polygon([P(p) for p in d["water_poly"]], fill=(150, 180, 215))
if len(d["river"]) >= 2:
    dr.line([P(p) for p in d["river"]], fill=(60, 100, 190), width=max(1, int(d["river_w"] * S)))
for q in d["parcels"]:
    if len(q[1]) >= 3:
        dr.polygon([P(p) for p in q[1]], outline=(190, 180, 150))
col = {"primary": (200, 120, 40), "street": (120, 120, 120), "lane": (170, 170, 170), "quay": (20, 20, 120), "bank": (40, 40, 160), "ringroad": (90, 90, 90)}
for e in d["edges"]:
    dr.line([P(e[2:4]), P(e[4:6])], fill=col.get(e[0], (0, 0, 0)), width=max(1, int(e[1] * S)))
for b in d["buildings"]:
    if len(b) >= 3:
        dr.polygon([P(p) for p in b], fill=(140, 70, 50))
if len(d["plaza"]) >= 3:
    dr.polygon([P(p) for p in d["plaza"]], outline=(0, 140, 0), width=3)
if len(d["ring"]) >= 3:
    dr.line([P(p) for p in d["ring"] + d["ring"][:1]], fill=(80, 80, 80), width=3)
x, y = P(d["market"]); dr.ellipse([x - 6, y - 6, x + 6, y + 6], outline=(220, 0, 0), width=3)
dr.rectangle([0, 0, W, 14], fill=(0, 0, 0))
dr.text((3, 1), (sys.argv[3] if len(sys.argv) > 3 else sys.argv[1])[:170] + f"  pop {d['pop']:.0f}/{d['target']:.0f}", fill=(255, 255, 255))
im.save(sys.argv[2])
