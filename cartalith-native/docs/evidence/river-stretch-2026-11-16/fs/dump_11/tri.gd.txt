extends SceneTree
func _init() -> void:
	for st in [0, 1]:
		var f := FileAccess.open("<evidence>/dump_11/246371_384_11_stretch%d.json" % st, FileAccess.READ)
		var d: Dictionary = JSON.parse_string(f.get_as_text())
		var wp := PackedVector2Array()
		for p in d["water_poly"]:
			wp.append(Vector2(p[0], p[1]))
		var tri := Geometry2D.triangulate_polygon(wp)
		# fan area from pts[0] (the fallback) vs. the polygon's own shoelace area
		var fan := 0.0
		for i in range(1, wp.size() - 1):
			fan += absf((wp[i] - wp[0]).cross(wp[i + 1] - wp[0])) / 2.0
		print("stretch %d: water_poly %d pts, triangulates %s, fan-fallback area %.0f m2 (%.1f ha)" % [st, wp.size(), str(not tri.is_empty()), fan, fan / 10000.0])
	quit()
