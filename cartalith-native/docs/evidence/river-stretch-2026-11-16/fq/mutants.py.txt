only = set(sys.argv[2:])
out = open(sys.argv[1], "a", encoding="utf-8")
for mid, f, old, new, crate, flt in M:
    if only and mid not in only:
        continue
    path = os.path.join(ROOT, f)
    src = open(path, encoding="utf-8", newline="").read()
    h0 = hashlib.sha1(src.encode()).hexdigest()
    o2, n2 = old, new
    if "\r\n" in src:
        o2, n2 = old.replace("\n", "\r\n"), new.replace("\n", "\r\n")
    n = src.count(o2)
    if n != 1:
        out.write(f"{mid}: NOT-ONCE ({n})\n"); out.flush(); continue
    try:
        open(path, "w", encoding="utf-8", newline="").write(src.replace(o2, n2))
        b = subprocess.run(["cargo", "test", "-p", crate, "--lib", "--no-run"], cwd=ROOT, env=ENV, capture_output=True, text=True)
        if b.returncode != 0:
            res = "BUILD_ERROR"
        else:
            t = subprocess.run(["cargo", "test", "-p", crate, "--lib", flt], cwd=ROOT, env=ENV, capture_output=True, text=True)
            res = "KILLED" if t.returncode != 0 else "SURVIVED"
            if res == "KILLED":
                fails = [l for l in t.stdout.splitlines() if l.startswith("test ") and l.endswith("FAILED")]
                res += " by " + ", ".join(x.split()[1] for x in fails[:4])
    finally:
        open(path, "w", encoding="utf-8", newline="").write(src)
    assert hashlib.sha1(open(path, encoding="utf-8", newline="").read().encode()).hexdigest() == h0
    out.write(f"{mid}: {res}\n"); out.flush()
