#!/bin/bash
# shots.sh: Ruling FQ frames. City Viewer opening frames before (--stretch 0) and after (--stretch 1, the app default),
# same DLL, app worlds at 384x288; then the map-level probe. Windowed, run under the window lock (withlock.sh).
GODOT="/c/Users/Vincent/AppData/Local/Microsoft/WinGet/Packages/GodotEngine.GodotEngine_Microsoft.Winget.Source_8wekyb3d8bbwe/Godot_v4.7.2-stable_win64_console.exe"
S=C:/cw/stretch2_scratch
cd C:/cw/stretch2/cartalith-native/godot-project
for run in 483920:0,2,4,6,11,12,13,15,17,23 912345:0,3,8,9,15 246371:2,6,11,15; do
  seed=${run%%:*}; ids=${run#*:}
  for st in 0 1; do
    APPDATA=$S/appdata timeout 2400 "$GODOT" --path . --resolution 1600x1000 _viewerframe_probe.tscn -- --seed $seed --grid 384x288 --stretch $st --shots $S/shots/s$st --ids $ids > $S/shots/${seed}_s$st.log 2>&1
    echo "$seed stretch $st exit $?"
  done
done
APPDATA=$S/appdata timeout 1800 "$GODOT" --path . --resolution 1600x1000 _stretchmap_probe.tscn -- --seed 483920 --grid 384x288 --out $S/shots/map > $S/shots/map.log 2>&1
echo "map exit $?"
