only = sys.argv[1:]
for name, path, old, new, cmd in M:
    if only and name not in only:
        continue
    raw = open(path, 'rb').read()
    crlf = b'\r\n' in raw
    text = raw.decode('utf-8')
    o, n = (old.replace('\n', '\r\n'), new.replace('\n', '\r\n')) if crlf else (old, new)
    c = text.count(o)
    if c != 1:
        print(f'{name}: SKIP (occurs {c}x)', flush=True); continue
    try:
        open(path, 'wb').write(text.replace(o, n).encode('utf-8'))
        r = subprocess.run(cmd, cwd=ROOT, capture_output=True, text=True, env=dict(os.environ, CARGO_TARGET_DIR='C:/cw/hr_target'))
        out = r.stdout + r.stderr
        if 'error[' in out or 'error: could not compile' in out:
            v = 'BUILD_ERROR'
        elif r.returncode == 0:
            v = 'SURVIVED'
        else:
            v = 'KILLED by ' + ','.join(sorted({l.split()[1] for l in out.splitlines() if l.startswith('test ') and l.endswith('FAILED')}))
        print(f'{name}: {v}', flush=True)
    finally:
        open(path, 'wb').write(raw)
