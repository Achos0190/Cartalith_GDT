+/// A block outline with its near-duplicate vertices (closer than 6 m) merged and its
+/// near-straight vertices (turn under 20 degrees) dropped: the corners a reader of
+/// the plan would count. A labelled judgement on both thresholds; 6 m is under the
+/// narrowest street, 20 degrees under the branch jitter's two sigma (2 x 15).
+fn ps_corners(poly: &[Vec2], turn_deg: f64) -> Vec<(f64, f64)> {
+    let mut p: Vec<(f64, f64)> = Vec::new();
+    for v in poly {
+        if p.last().is_none_or(|l: &(f64, f64)| (l.0 - v.x).hypot(l.1 - v.y) >= 6.0) {
+            p.push((v.x, v.y));
+        }
+    }
+    while p.len() > 2 && { let (a, b) = (p[0], p[p.len() - 1]); (a.0 - b.0).hypot(a.1 - b.1) < 6.0 } {
+        p.pop();
+    }
+    loop {
+        let n = p.len();
+        if n <= 3 {
+            return p;
+        }
+        let mut drop = None;
+        let mut least = f64::INFINITY;
+        for i in 0..n {
+            let (a, b, c) = (p[(i + n - 1) % n], p[i], p[(i + 1) % n]);
+            let t1 = (b.1 - a.1).atan2(b.0 - a.0);
+            let t2 = (c.1 - b.1).atan2(c.0 - b.0);
+            let mut d = (t2 - t1).abs() % (2.0 * std::f64::consts::PI);
+            if d > std::f64::consts::PI {
+                d = 2.0 * std::f64::consts::PI - d;
+            }
+            if d < least {
+                least = d;
+                drop = Some(i);
+            }
+        }
+        if least.to_degrees() < turn_deg {
+            p.remove(drop.unwrap());
+        } else {
+            return p;
+        }
+    }

+/// Interior angles (degrees, 0..180) at each corner of a simplified outline.
+fn ps_interior(p: &[(f64, f64)]) -> Vec<f64> {
+    let n = p.len();
+    (0..n)
+        .map(|i| {
+            let (a, b, c) = (p[(i + n - 1) % n], p[i], p[(i + 1) % n]);
+            let (u, v) = ((a.0 - b.0, a.1 - b.1), (c.0 - b.0, c.1 - b.1));

+/// Minimum-area bounding rectangle over the outline's edge directions: (long side, short side).
+/// The short side is the block's depth across: a lot of depth d needs a block about 2 d / 0.96
+/// across with lots from both long sides (`build_parcels_with`'s WAIST 0.48).
+fn ps_mabr(poly: &[Vec2]) -> (f64, f64) {
+    let mut best = (f64::INFINITY, 0.0, 0.0);
+    let n = poly.len();
+    for i in 0..n {
+        let (a, b) = (poly[i], poly[(i + 1) % n]);
+        let th = (b.y - a.y).atan2(b.x - a.x);
+        let (c, s) = (th.cos(), th.sin());
+        let (mut x0, mut x1, mut y0, mut y1) = (f64::INFINITY, f64::NEG_INFINITY, f64::INFINITY, f64::NEG_INFINITY);
+        for v in poly {
+            let (x, y) = (v.x * c + v.y * s, -v.x * s + v.y * c);
+            x0 = x0.min(x);
+            x1 = x1.max(x);
+            y0 = y0.min(y);
+            y1 = y1.max(y);
+        }
+        let (w, h) = (x1 - x0, y1 - y0);
+        if w * h < best.0 {
+            best = (w * h, w.max(h), w.min(h));
+        }
+    }

+    fn acc(sum: &mut (f64, f64, f64), a: Vec2, b: Vec2, w: f64) {
+        let th = js_atan2(b.y - a.y, b.x - a.x);
+        sum.0 += w * (2.0 * th).cos();
+        sum.1 += w * (2.0 * th).sin();
+        sum.2 += w;
+    }
+    impl Field {
+        /// (axis bearing, strength 0..1) at `p`, `d_m` its market distance. A doubled-angle
+        /// (orientation tensor) blend: approach roads within ~80 m, the river within ~80 m (x1.5),
+        /// the market spine (the primary through the market) out to ~250 m; strength is the blend's
+        /// coherence times its total weight (capped at 1), times a ramp from 0 inside the organic
+        /// core (`core` = 0.38 max_rf) to 1 at `core + ramp`, times `gain`.
+        pub fn at(&self, p: Vec2, d_m: f64) -> (f64, f64) {
+            let mut s = (0.0, 0.0, 0.0);
+            for &(a, b) in &self.prims {
+                let d = dist_pt_seg(p, a, b);
+                acc(&mut s, a, b, 1.0 / (1.0 + (d / 80.0).powi(2)));
+            }
+            if self.river.len() >= 2 {
+                let (mut bd, mut bi) = (f64::INFINITY, 0);
+                for i in 0..self.river.len() - 1 {
+                    let d = dist_pt_seg(p, self.river[i], self.river[i + 1]);
+                    if d < bd {
+                        bd = d;
+                        bi = i;
+                    }
+                }
+                acc(&mut s, self.river[bi], self.river[bi + 1], 1.5 / (1.0 + (bd / 80.0).powi(2)));
+            }
+            if let Some((a, b)) = self.spine {
+                let (dx, dy) = (b.x - a.x, b.y - a.y);
+                let ll = (dx * dx + dy * dy).sqrt().max(1e-9);
+                let d = ((p.x - a.x) * dy - (p.y - a.y) * dx).abs() / ll;
+                acc(&mut s, a, b, 0.7 / (1.0 + (d / 250.0).powi(2)));
+            }
+            if s.2 <= 0.0 {
+                return (0.0, 0.0);
+            }
+            let mag = (s.0 * s.0 + s.1 * s.1).sqrt();
+            let coh = mag / s.2;
+            let th = 0.5 * s.1.atan2(s.0);
+            let ramp = ((d_m - self.core) / self.ramp).clamp(0.0, 1.0);
+            (th, (coh * s.2.min(1.0) * ramp * self.gain).clamp(0.0, 1.0))
+        }
+    }
+    /// Signed difference a - b wrapped to (-pi, pi].
+    pub fn wrap(d: f64) -> f64 {
+        let mut x = d % (2.0 * PI);
+        if x > PI {
+            x -= 2.0 * PI;
+        }
+        if x <= -PI {
+            x += 2.0 * PI;
+        }
+        x
+    }
+    /// The field axis (one of four) nearest heading `ang`.

+            let d = wrap(ang - c).abs();
+            if d < bd {
+                bd = d;
+                best = c;
+            }
+        }
+        best
+    }
