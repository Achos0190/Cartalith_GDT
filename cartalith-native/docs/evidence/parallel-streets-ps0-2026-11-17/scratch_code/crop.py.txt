import sys
from PIL import Image, ImageDraw, ImageFont
# crop.py img x0 y0 x1 y1 out [scale] : grid every 50 original px, labelled in original px
img,x0,y0,x1,y1,out=sys.argv[1],*map(int,sys.argv[2:6]),sys.argv[6]
sc=float(sys.argv[7]) if len(sys.argv)>7 else 1.0
im=Image.open(img).convert('RGB').crop((x0,y0,x1,y1))
im=im.resize((int((x1-x0)*sc),int((y1-y0)*sc)),Image.LANCZOS)
d=ImageDraw.Draw(im)
for gx in range((x0//50+1)*50,x1,50):
    X=(gx-x0)*sc; d.line([(X,0),(X,im.size[1])],fill=(0,160,255) if gx%100 else (255,0,255),width=1)
    if gx%100==0: d.text((X+2,2),str(gx),fill=(255,0,255))
for gy in range((y0//50+1)*50,y1,50):
    Y=(gy-y0)*sc; d.line([(0,Y),(im.size[0],Y)],fill=(0,160,255) if gy%100 else (255,0,255),width=1)
    if gy%100==0: d.text((2,Y+2),str(gy),fill=(255,0,255))
im.save(out)
