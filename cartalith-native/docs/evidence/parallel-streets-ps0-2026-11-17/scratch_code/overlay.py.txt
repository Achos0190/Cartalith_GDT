import sys, ast
sys.path.insert(0, 'C:/cw/ps0_scratch/trace')
from plan_measure import measure, summary
from PIL import Image, ImageDraw
t = ast.literal_eval(open(sys.argv[1]).read())
m = measure(t, t['m_per_px'])
im = Image.open(t['image']).convert('RGB')
x0, y0, x1, y1 = t['crop']
d = ImageDraw.Draw(im, 'RGBA')
col = {3: (255, 0, 0, 90), 4: (0, 200, 0, 70)}
for b in m['blocks']:
    d.polygon([tuple(p) for p in b['poly']], fill=col.get(b['corners'], (0, 80, 255, 70)), outline=(0, 0, 0, 255))
for s in t['streets']:
    d.line([tuple(p) for p in s['pts']], fill=(255, 0, 255, 200), width=2)
im = im.crop((x0, y0, x1, y1))
sc = float(sys.argv[3]) if len(sys.argv) > 3 else 1.0
if sc != 1.0:
    im = im.resize((int(im.size[0]*sc), int(im.size[1]*sc)))
im.save(sys.argv[2])
print(summary(t['name'], m))
