ENV = dict(os.environ, CARGO_TARGET_DIR=r"C:/cw/townleft_mut_target")

sel = [int(a) for a in sys.argv[1:]] or range(len(M))
for k in sel:
    f, old, new = M[k]
    p = SRC + f
    orig = open(p, "rb").read()
    h = hashlib.sha256(orig).hexdigest()
    s = orig.decode("utf-8")
    assert s.count(old) == 1, (k, f, s.count(old))
    try:
        open(p, "wb").write(s.replace(old, new).encode("utf-8"))
        r = subprocess.run(["cargo", "test", "--release", "-p", "cartalith-urban", "--lib"], cwd=ROOT, env=ENV, capture_output=True, text=True, encoding="utf-8", errors="replace")
        out = r.stdout + r.stderr
        if "error[" in out or "could not compile" in out and "test result" not in out:
            v = "BUILD_ERROR"
        elif r.returncode == 0 or all(n == "site::tests::golden_build_site" for n in (l.split()[1] for l in out.splitlines() if l.startswith("test ") and l.endswith("FAILED"))):
            # golden_build_site fails on origin/main in release (a NaN sign bit), so it alone is no kill.
            v = "SURVIVED"
        else:
            v = "KILLED " + ",".join(n for n in (l.split()[1] for l in out.splitlines() if l.startswith("test ") and l.endswith("FAILED")) if n != "site::tests::golden_build_site")[:300]
    finally:
        open(p, "wb").write(orig)
        assert hashlib.sha256(open(p, "rb").read()).hexdigest() == h
    print(f"{k:2d} {f} {v} | {old.strip()[:70]!r} -> {new.strip()[:50]!r}", flush=True)
