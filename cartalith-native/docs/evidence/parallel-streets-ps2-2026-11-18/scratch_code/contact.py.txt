"""PS-2 step 0 (scratch): per seed a contact sheet (rows = the scope's towns; columns = City Viewer opening frame OFF, ON,
then x4 OFF, ON), a full-size pair per town and zoom, and needle crops (lots of the PS-0 section 8.4 definition red)
from the probe's layout dumps, the same 260 m window off and on."""
import re, json, math
from PIL import Image, ImageDraw
S = 'C:/cw/ps2b_scratch/shots'; O = 'C:/cw/ps2b_scratch/ev'
TW = 420
W_FLOOR, R95, R99 = 4.15, 3.5, 6.0
def needle(q):
    if len(q) != 4: return True
    d = lambda a, b: math.hypot(a[0]-b[0], a[1]-b[1])
    f, b = d(q[0], q[1]), d(q[3], q[2]); dep = (d(q[0], q[3])+d(q[1], q[2]))/2
    wmin = min(f, b); r = dep/max((f+b)/2, 1e-6)
    return (wmin < W_FLOOR and r > R95) or r > R99 or wmin < 1.0
def crop(lay, cx, cy, title, WIN=260.0, PX=3.0):
    Sz = int(WIN*PX); im = Image.new('RGB', (Sz, Sz+22), (236, 230, 214)); d = ImageDraw.Draw(im)
    T = lambda p: ((p[0]-cx+WIN/2)*PX, (p[1]-cy+WIN/2)*PX+22)
    n = nn = 0
    for cls, q in lay.get('parcels', []):
        if not q: continue
        c = (sum(x for x, _ in q)/len(q), sum(y for _, y in q)/len(q))
        if abs(c[0]-cx) > WIN/2 or abs(c[1]-cy) > WIN/2: continue
        bad = needle(q); n += 1; nn += bad
        d.polygon([T(p) for p in q], fill=(214, 60, 50) if bad else (222, 205, 170), outline=(150, 130, 100))
    for b in lay.get('buildings', []):
        if len(b) >= 3: d.polygon([T(p) for p in b], fill=(70, 60, 55))
    for e in lay.get('edges', []):
        d.line([T((e[2], e[3])), T((e[4], e[5]))], fill=(120, 120, 120), width=max(1, int(e[1]*PX*0.6)))
    d.rectangle([0, 0, Sz, 21], fill=(255, 255, 255)); d.text((4, 5), f'{title}: needle lots {nn} of {n} ({100*nn/max(n,1):.1f} %)', fill=(0, 0, 0))
    return im
for seed in ('483920', '912345', '246371'):
    log = open(f'{S}/{seed}.log', encoding='utf-8', errors='replace').read()
    rows = re.findall(r'PARALLEL1 #(\d+) buildings (\d+) edges (\d+) type (\S+)', log)
    off = dict((r[0], r[1:3]) for r in re.findall(r'PARALLEL0 #(\d+) buildings (\d+) edges (\d+)', log))
    sheet_rows = []
    for r in rows:
        tid = r[0]; ims = []
        for z in ('z1', 'z4'):
            a = Image.open(f'{S}/viewer_{seed}_{tid}_parallel0_{z}.png').convert('RGB'); b = Image.open(f'{S}/viewer_{seed}_{tid}_parallel1_{z}.png').convert('RGB')
            P = Image.new('RGB', (a.size[0]+b.size[0]+10, max(a.size[1], b.size[1])+26), (255, 255, 255))
            P.paste(a, (0, 26)); P.paste(b, (a.size[0]+10, 26))
            ImageDraw.Draw(P).text((4, 6), f'seed {seed} #{tid} (384 x 288) {z}: OFF (left, buildings {off[tid][0]}, edges {off[tid][1]}) | ON, PS-2 step-0 prototype (right, buildings {r[1]}, edges {r[2]}); type {r[3]}; scratch', fill=(0, 0, 0))
            P.save(f'{O}/frames/pair_{seed}_{tid}_{z}.png'); ims += [a, b]
        lo = json.load(open(f'{S}/dump_{seed}_{tid}_p0.json')); ln = json.load(open(f'{S}/dump_{seed}_{tid}_p1.json'))
        # the window: 260 m round the mean of the on layout's lots within 300 m of its market, the same for off
        m = ln['market']
        cs = [(sum(x for x, _ in q)/len(q), sum(y for _, y in q)/len(q)) for _, q in ln['parcels'] if q]
        cs = [c for c in cs if math.hypot(c[0]-m[0], c[1]-m[1]) < 300] or [tuple(m)]
        cx, cy = sum(c[0] for c in cs)/len(cs), sum(c[1] for c in cs)/len(cs)
        a = crop(lo, cx, cy, f'{seed} #{tid} OFF'); b = crop(ln, cx, cy, f'{seed} #{tid} ON')
        P = Image.new('RGB', (a.size[0]*2+10, a.size[1]), (255, 255, 255)); P.paste(a, (0, 0)); P.paste(b, (a.size[0]+10, 0))
        P.save(f'{O}/needles/needles_{seed}_{tid}.png')
        sheet_rows.append((r, ims))
    th = int(TW*sheet_rows[0][1][0].size[1]/sheet_rows[0][1][0].size[0])
    sheet = Image.new('RGB', (4*TW+60, 30+len(sheet_rows)*(th+24)), (255, 255, 255)); d = ImageDraw.Draw(sheet)
    d.text((6, 8), f'seed {seed}, 384 x 288: PS-2 step 0 (scratch prototype, NO-GO). Columns: opening frame OFF, ON; x4 (market-centred) OFF, ON.', fill=(0, 0, 0))
    for k, (r, ims) in enumerate(sheet_rows):
        y = 30+k*(th+24)
        d.text((6, y), f'#{r[0]} type {r[3]}: off buildings {off[r[0]][0]} edges {off[r[0]][1]}; on buildings {r[1]} edges {r[2]}', fill=(0, 0, 0))
        for j, im in enumerate(ims):
            t = im.copy(); t.thumbnail((TW, th)); sheet.paste(t, (6+j*(TW+14), y+16))
    sheet.save(f'{O}/contact_{seed}.png'); print(seed, sheet.size, len(sheet_rows))
