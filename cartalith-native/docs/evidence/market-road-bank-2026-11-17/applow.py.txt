"""applow.py <dir> <gw>: per app-world town (dumped by _viewerframe_probe.gd --dump --ids -1), buildings per 1 000 of
target, against the median of all; lists every town below 0.6 of the median."""
import json, glob, sys, statistics, re
rows = []
for f in sorted(glob.glob(f"{sys.argv[1]}/*_{sys.argv[2]}_*.json")):
    m = re.search(r"(\d+)_(\d+)_(\d+)\.json$", f)
    d = json.load(open(f))
    rows.append((int(m.group(1)), int(m.group(3)), d["site_kind"], d["target"], len(d["buildings"]), d["pop"]))
per = [r[4] / r[3] * 1000 for r in rows]
med = statistics.median(per)
print(f"{len(rows)} towns >= 1500 at {sys.argv[2]}: buildings per 1 000 of target median {med:.0f}")
for r, p in sorted(zip(rows, per), key=lambda x: x[1]):
    flag = "  <-- under 0.6" if p < 0.6 * med else ""
    print(f"  seed {r[0]} #{r[1]:<3} {r[2]:12s} target {r[3]:6.0f} bld {r[4]:5d} housed {100*r[5]/r[3]:5.1f} % {p:4.0f}/1000 ({p/med:.2f}){flag}")
