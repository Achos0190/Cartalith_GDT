"""foot.py <off_dir> <on_dir>: per-town settled area on / off (both_banks_audit 'plots #i' lines, TOWN_RIVER_LIST=1),
paired by seed block and town id; towns >= 1500 and >= 5000; also housed % on / off."""
import re, sys, statistics as st
def rd(d, gw):
    t = open(f"{d}/both_banks_audit_{gw}.txt", encoding="utf-8", errors="replace").read()
    # Paired by order of appearance: both runs lay out the same settlements in the same order (the seeds
    # print no header, and ids repeat per seed).
    out = {}
    for n, m in enumerate(re.finditer(r"plots #(\d+) target (\d+) housed (\d+) settled ([\d.]+) ha", t)):
        out[(n, int(m.group(1)))] = (int(m.group(2)), int(m.group(3)), float(m.group(4)))
    return out
for gw in (384, 2048):
    a, b = rd(sys.argv[1], gw), rd(sys.argv[2], gw)
    keys = sorted(set(a) & set(b))
    for lo in (1500, 5000):
        r = [b[k][2] / a[k][2] for k in keys if a[k][0] >= lo and a[k][2] > 0]
        h = [(b[k][1] - a[k][1]) / a[k][0] * 100 for k in keys if a[k][0] >= lo]
        q = sorted(r)
        print(f"{gw} towns>={lo} n {len(r)}: settled on/off min {q[0]:.2f} p10 {q[len(q)//10]:.2f} median {st.median(q):.2f} p90 {q[(len(q)*9)//10]:.2f} max {q[-1]:.2f}; housed change median {st.median(h):+.1f} pts of target")
