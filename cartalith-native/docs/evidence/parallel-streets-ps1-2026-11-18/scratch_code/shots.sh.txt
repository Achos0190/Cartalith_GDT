#!/bin/bash
# shots.sh: PS-1 frames. Per seed, the scope's 14 towns (burgage-backlanes-2026-11-17/frames, found by map cell): the City
# Viewer's opening frame with urban_parallel_streets off, then on live in the same session with the field drawn
# (--parallel 2). Windowed, scratch APPDATA, under the shared window lock (released only by the run that made it).
G="/c/Users/Vincent/AppData/Local/Microsoft/WinGet/Packages/GodotEngine.GodotEngine_Microsoft.Winget.Source_8wekyb3d8bbwe/Godot_v4.7.2-stable_win64_console.exe"
O=C:/cw/ps1_scratch/shots; mkdir -p $O C:/cw/ps1_scratch/appdata
ME="ps1 agent (PS-1 field frames, 2026-11-18) pid $$"
until mkdir C:/cw/.win_lock 2>/dev/null; do sleep 15; done
echo "$ME started $(date)" > C:/cw/.win_lock/owner.txt
cd C:/cw/ps1/cartalith-native/godot-project
declare -A CELLS=( [483920]="10,54;142,235;120,220;152,262;77,195" [912345]="312,235;375,180;81,105;92,121" [246371]="240,205;96,92;177,98;314,122;216,193" )
for seed in 483920 912345 246371; do
  APPDATA=C:/cw/ps1_scratch/appdata timeout 3000 "$G" --path . --resolution 1600x1000 _viewerframe_probe.tscn -- --seed $seed --grid 384x288 --quick 1 --parallel 2 --shots $O --cells "${CELLS[$seed]}" > $O/${seed}.log 2>&1
  echo "$seed rc $?"
done
grep -q "ps1 agent (PS-1 field frames, 2026-11-18) pid $$" C:/cw/.win_lock/owner.txt 2>/dev/null && rm -rf C:/cw/.win_lock
echo SHOTSDONE
