+        let opts = crate::growth::GrowOpts { max_rf: 720.0, field, ..Default::default() };
+        let ws = crate::growth::WallState::default();
+        crate::growth::grow_far_bank(7, &site, &anchors, &mut g, &ws, &opts, 8, 1.0, Vec2::new(800.0, r + 30.0));
+        let (n1, _) = bank_lengths(&g);
+        assert!((n1 - n0).abs() < 1e-6, "the market's bank changed: {n0} -> {n1}");
+        let (mut tot, mut on) = (0.0, 0.0);
+        for e in g.edges[before..].iter().filter(|e| e.alive) {
+            let (a, b) = (g.nodes[e.a].pt(), g.nodes[e.b].pt());
+            if north(a.lerp(b, 0.5)) {
+                continue;
+            }
+            let l = a.dist(b);
+            tot += l;
+            if crate::field::axis_off((b.y - a.y).atan2(b.x - a.x), 0.0) <= 0.35 {
+                on += l;
+            }
+        }
+        on / tot
+    };
+    let river = Vec::from([Vec2::new(0.0, r), Vec2::new(1700.0, r)]);
+    let grain = crate::field::DirectionField { sources: vec![crate::field::Source { kind: crate::field::SourceKind::River, line: river, weight: 1.5, reach_m: 400.0 }], core_m: 0.0, ramp_m: 0.0 };
+    let (without, with) = (on_axis(None), on_axis(Some(grain)));
+    assert!((without - 0.7608).abs() < 1e-4, "without the grain {without}");
+    assert!((with - 1.0).abs() < 1e-9, "with the grain {with}");
