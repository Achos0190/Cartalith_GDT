#!/bin/bash
# PS-4 frames: City Viewer, the shell's world at 384 x 288, two sessions per seed (each off, then on live):
#   mid   = PS-2 as shipped (every PS-4 piece disabled by scratch knobs)
#   after = PS-4 (the base's stretch choice; the other pieces' knobs off, as measured)
# Holds C:/cw/.win_lock (MISTAKES #26): acquired with mkdir in an until-loop, owner file written at once, removed only
# when the owner file is ours.
G="/c/Users/Vincent/AppData/Local/Microsoft/WinGet/Packages/GodotEngine.GodotEngine_Microsoft.Winget.Source_8wekyb3d8bbwe/Godot_v4.7.2-stable_win64_console.exe"
S=C:/cw/ps4_scratch
L=C:/cw/.win_lock
ME="ps4-frames $$ $(date +%s)"
until mkdir "$L" 2>/dev/null; do sleep 20; done
echo "$ME" > "$L/owner_ps4"
echo "lock taken: $ME"
cd C:/cw/ps4/cartalith-native/godot-project
declare -A IDS=( [483920]="0,3,4,6,16,5,13,2" [912345]="0,1,2,4,6,20,8,32" )
for seed in 483920 912345; do
  PS4_A=0 PS4_B=0 PS4_C=0 PS4_E=0 APPDATA=$S/appdata timeout 3000 "$G" --path . --resolution 1600x1000 _ps4_probe.tscn -- --seed $seed --grid 384x288 --quick 1 --parallel 2 --shots $S/shots_mid --ids "${IDS[$seed]}" > $S/mid_${seed}.log 2>&1
  echo "$seed mid rc $?"
  PS4_A=0 PS4_B=0 PS4_C=0 APPDATA=$S/appdata timeout 3000 "$G" --path . --resolution 1600x1000 _ps4_probe.tscn -- --seed $seed --grid 384x288 --quick 1 --parallel 2 --shots $S/shots_after --ids "${IDS[$seed]}" > $S/after_${seed}.log 2>&1
  echo "$seed after rc $?"
done
if [ "$(cat "$L/owner_ps4" 2>/dev/null)" = "$ME" ]; then rm -rf "$L"; echo "lock released"; else echo "lock not ours, left alone"; fi
echo SHOTSDONE
