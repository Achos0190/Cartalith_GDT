#!/bin/bash
# var.sh <exe> <outdir> <gw:gh> [env...]: the switch-on audits only (parallel_streets_audit off/on CSVs, both_banks and stretch_water with the switch on).
EXE=$1; OUT=$2; sz=$3; gw=${sz%%:*}; gh=${sz##*:}; shift 3
cd /c/cw/ps4/cartalith-native/crates/cartalith-godot
mkdir -p $OUT/$gw
env "$@" TOWN_RIVER_GW=$gw TOWN_RIVER_GH=$gh TOWN_RIVER_VILLAGES=1 PS1_CSV=$OUT/$gw $EXE parallel_streets_audit --ignored --nocapture --test-threads 1 > $OUT/audit_$gw.txt 2>&1
python C:/cw/ps4_scratch/ps1_agg.py $OUT/$gw $gw > $OUT/agg_$gw.txt 2>&1
env "$@" TOWN_RIVER_GW=$gw TOWN_RIVER_GH=$gh TOWN_RIVER_VILLAGES=1 TOWN_RIVER_PARALLEL=1 $EXE both_banks_audit --ignored --nocapture --test-threads 1 > $OUT/bb_${gw}_p1.txt 2>&1
env "$@" TOWN_RIVER_GW=$gw TOWN_RIVER_GH=$gh TOWN_RIVER_VILLAGES=1 TOWN_RIVER_PARALLEL=1 $EXE stretch_water_audit --ignored --nocapture --test-threads 1 > $OUT/sw_${gw}_p1.txt 2>&1
echo done $OUT $gw
